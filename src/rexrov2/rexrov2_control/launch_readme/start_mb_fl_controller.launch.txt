<!-- 
启动RexROV2水下机器人的[模型基于反馈线性化控制器](https://github.com/uuvsimulator/uuv_simulator/blob/master/uuv_control/uuv_trajectory_control/scripts/rov_mb_fl_controller.py)
该控制器的默认参数Kp、Kd和Ki使用[SMAC](https://github.com/automl/SMAC3)算法进行了优化

使用示例：

1. 启动Gazebo海洋环境仿真：
roslaunch uuv_gazebo_worlds ocean_waves.launch

2. 加载RexROV2机器人模型：
roslaunch rexrov2_description upload_rexrov2.launch

3. 启动反馈线性化控制器：
roslaunch rexrov2_control start_mb_fl_controller.launch
-->
<launch>
    <!-- 基础配置参数 -->
    <arg name="uuv_name" default="rexrov2"/> <!-- 机器人命名空间 -->
    <!-- Flag to start recoding a rosbag  -->
    <arg name="record" default="false" doc="Record ROS bag"/> <!-- 是否记录ROS数据包 -->

    <!-- 可视化与坐标系配置 -->
    <arg name="gui_on" default="true"/> <!-- 是否启动Gazebo客户端和RViz -->
    <arg name="use_ned_frame" default="false"/> <!-- 坐标系选择：false=ENU, true=NED -->

    <!-- 反馈线性化控制器PID参数 -->
    <!-- 比例增益矩阵，对应6个自由度：x,y,z,roll,pitch,yaw -->
    <arg name="Kp" default="7750.307,7750.307,7750.307,7430.825,7430.825,7430.825"/>
    <!-- 微分增益矩阵，对应6个自由度 -->
    <arg name="Kd" default="8953.924,8953.924,8953.924,8267.137,8267.137,8267.137"/>
    <!-- 积分增益矩阵，对应6个自由度 -->
    <arg name="Ki" default="9144.575,9144.575,9144.575,2095.077,2095.077,2095.077"/>

    <!-- 遥操作配置参数 -->
    <arg name="teleop_on" default="false"/> <!-- 是否启用手柄控制 -->
    <arg name="joy_id" default="0"/> <!-- 手柄设备ID -->
    <!-- 手柄轴映射配置 -->
    <arg name="axis_yaw" default="0"/> <!-- 偏航角控制轴 -->
    <arg name="axis_x" default="4"/> <!-- X轴速度控制轴 -->
    <arg name="axis_y" default="3"/> <!-- Y轴速度控制轴 -->
    <arg name="axis_z" default="1"/> <!-- Z轴速度控制轴 -->

    <!-- 控制器模型参数文件 -->
    <!-- 包含机器人水动力学参数和模型信息的配置文件 -->
    <arg name="model_params_file" default="$(find rexrov2_control)/config/model_params.yaml"/>

    <!-- 推力器管理器配置 -->
    <!-- 推力器管理器生成文件的输出目录 -->
    <arg name="thruster_manager_output_dir" default="$(find rexrov2_control)/config"/>
    <!-- 推力器管理器配置文件路径 -->
    <arg name="thruster_manager_config_file" default="$(find rexrov2_control)/config/thruster_manager.yaml"/>
    <!-- 推力器分配矩阵文件路径 -->
    <arg name="tam_file" default="$(find rexrov2_control)/config/TAM.yaml"/>

    <!-- 主控制器启动：模型基于反馈线性化控制器 -->
    <include file="$(find uuv_trajectory_control)/launch/rov_mb_fl_controller.launch">
      <!-- 机器人命名空间 -->
      <arg name="uuv_name" value="$(arg uuv_name)"/>
      <!-- 可视化开关 -->
      <arg name="gui_on" value="$(arg gui_on)"/>
      <!-- 控制输出饱和度限制 -->
      <arg name="saturation" value="2000"/>
      <!-- 控制器增益参数 -->
      <arg name="Kp" value="$(arg Kp)"/> <!-- 比例增益 -->
      <arg name="Kd" value="$(arg Kd)"/> <!-- 微分增益 -->
      <arg name="Ki" value="$(arg Ki)"/> <!-- 积分增益 -->
      <!-- 坐标系选择 -->
      <arg name="use_ned_frame" value="$(arg use_ned_frame)"/>
      <!-- 推力器管理相关配置 -->
      <arg name="thruster_manager_output_dir" value="$(arg thruster_manager_output_dir)"/>
      <arg name="thruster_manager_config_file" value="$(arg thruster_manager_config_file)"/>
      <arg name="tam_file" value="$(arg tam_file)"/>
      <!-- 机器人模型参数文件 -->
      <arg name="model_params_file" value="$(arg model_params_file)"/>
    </include>

    <!-- 数据记录功能 -->
    <!-- Initialize the recording fo the simulation according to the record flag  -->
    <include file="$(find rexrov2_control)/launch/record_demo.launch">
      <arg name="record" value="$(arg record)"/> <!-- 记录开关 -->
      <arg name="use_ned_frame" value="$(arg use_ned_frame)"/> <!-- 坐标系 -->
      <arg name="uuv_name" value="$(arg uuv_name)"/> <!-- 机器人名称 -->
    </include>

    <!-- 遥操作功能 -->
    <group if="$(arg teleop_on)">
      <!-- Start joystick teleop node  -->
      <include file="$(find uuv_teleop)/launch/uuv_teleop.launch">
          <arg name="uuv_name" value="$(arg uuv_name)"/> <!-- 机器人命名空间 -->
          <arg name="joy_id" value="$(arg joy_id)"/> <!-- 手柄设备ID -->
          <arg name="output_topic" value="cmd_vel"/> <!-- 输出控制指令的话题 -->
          <arg name="message_type" value="twist"/> <!-- 消息类型：速度指令 -->
          <!-- 手柄轴映射 -->
          <arg name="axis_yaw" value="$(arg axis_yaw)"/> <!-- 偏航轴 -->
          <arg name="axis_x" value="$(arg axis_x)"/> <!-- X轴 -->
          <arg name="axis_y" value="$(arg axis_y)"/> <!-- Y轴 -->
          <arg name="axis_z" value="$(arg axis_z)"/> <!-- Z轴 -->
          <!-- 控制增益设置 -->
          <arg name="gain_yaw" default="0.2"/> <!-- 偏航控制增益 -->
          <arg name="gain_x" default="0.5"/> <!-- X轴控制增益 -->
          <arg name="gain_y" default="0.5"/> <!-- Y轴控制增益 -->
          <arg name="gain_z" default="0.5"/> <!-- Z轴控制增益 -->
      </include>
    </group>

    <!-- 可视化功能 -->
    <group if="$(arg gui_on)">
      <!-- Start RViz  -->
      <node name="rviz" pkg="rviz" type="rviz" output="screen" args="-d $(find rexrov2_control)/rviz/rexrov2_control.rviz"/>
      <!-- 启动RViz并加载预设的RexROV2控制配置文件 -->
    </group>
</launch>

<!-- 
启动RexROV2水下机器人的[6自由度PID控制器](https://github.com/uuvsimulator/uuv_simulator/blob/master/uuv_control/uuv_trajectory_control/scripts/rov_pid_controller.py)
该控制器的默认参数Kp、Kd和Ki使用[SMAC](https://github.com/automl/SMAC3)算法进行了优化

使用示例：

1. 启动Gazebo海洋环境仿真：
roslaunch uuv_gazebo_worlds ocean_waves.launch

2. 加载RexROV2机器人模型：
roslaunch rexrov2_description upload_rexrov2.launch

3. 启动PID控制器：
roslaunch rexrov2_control start_pid_controller.launch
-->
<launch>
    <!-- 基础配置参数 -->
    <arg name="uuv_name" default="rexrov2" doc="Namespace of the vehicle to be controller"/> <!-- 被控制机器人的命名空间 -->

    <!-- 数据记录与可视化配置 -->
    <!-- Flag to start recoding a rosbag  -->
    <arg name="record" default="false" doc="Record ROS bag"/> <!-- 是否记录ROS数据包，用于后续分析和调试 -->
    
    <arg name="gui_on" default="true" doc="Start Gazebo client and RViz"/> <!-- 是否启动Gazebo客户端和RViz可视化界面 -->
    <arg name="use_ned_frame" default="false" doc="If true, the reference frame is set to be NED (North-East-Down) instead of Gazebo's default ENU (East-North-Up)"/> <!-- 坐标系选择：false=ENU(东-北-天), true=NED(北-东-地) -->

    <!-- PID控制器增益参数 -->
    <!-- 比例增益矩阵，对应6个自由度：surge(x), sway(y), heave(z), roll(φ), pitch(θ), yaw(ψ) -->
    <arg name="Kp" default="11993.888,11993.888,11993.888,19460.069,19460.069,19460.069" doc="Coefficients for the $K_p$ diagonal matrix"/>
    <!-- 微分增益矩阵，提供阻尼抑制振荡 -->
    <arg name="Kd" default="9077.459,9077.459,9077.459,18880.925,18880.925,18880.925" doc="Coefficients for the $K_d$ diagonal matrix"/>
    <!-- 积分增益矩阵，消除稳态误差 -->
    <arg name="Ki" default="321.417,321.417,321.417,2096.951,2096.951,2096.951" doc="Coefficients for the $K_i$ diagonal matrix"/>

    <!-- 遥操作配置参数 -->
    <arg name="teleop_on" default="false" doc="Start joystick teleop node"/> <!-- 是否启用手柄遥操作功能 -->
    <arg name="joy_id" default="0" doc="ID of the joystick device"/> <!-- 手柄设备ID，支持多手柄配置 -->
    <!-- 手柄轴映射配置 -->
    <arg name="axis_yaw" default="0" doc="Joystick mapping for yaw angle input"/> <!-- 偏航角控制轴映射 -->
    <arg name="axis_x" default="4" doc="Joystick mapping for X component of velocity reference input"/> <!-- X轴(前进方向)速度控制轴映射 -->
    <arg name="axis_y" default="3" doc="Joystick mapping for Y component of velocity reference input"/> <!-- Y轴(横向)速度控制轴映射 -->
    <arg name="axis_z" default="1" doc="Joystick mapping for Z component of velocity reference input"/> <!-- Z轴(垂向)速度控制轴映射 -->

    <!-- 推力器管理器配置 -->
    <!-- 推力器管理器生成文件的输出目录 -->
    <arg name="thruster_manager_output_dir" default="$(find rexrov2_control)/config" doc="Output directory for the generated thruster manager"/>
    <!-- 推力器管理器配置文件路径 -->
    <arg name="thruster_manager_config_file" default="$(find rexrov2_control)/config/thruster_manager.yaml" doc="File storing the configuration of the thruster manager"/>
    <!-- 推力器分配矩阵文件路径 -->
    <arg name="tam_file" default="$(find rexrov2_control)/config/TAM.yaml" doc="File storing the thruster allocation matrix"/>

    <!-- 主控制器启动：6自由度PID控制器 -->
    <include file="$(find uuv_trajectory_control)/launch/rov_pid_controller.launch">
      <!-- 机器人命名空间 -->
      <arg name="uuv_name" value="$(arg uuv_name)"/>
      <!-- 可视化开关 -->
      <arg name="gui_on" value="$(arg gui_on)"/>
      <!-- 控制输出饱和度限制，防止推力器过载 -->
      <arg name="saturation" value="2000"/>
      <!-- PID控制器增益参数 -->
      <arg name="Kp" value="$(arg Kp)"/> <!-- 比例增益 -->
      <arg name="Kd" value="$(arg Kd)"/> <!-- 微分增益 -->
      <arg name="Ki" value="$(arg Ki)"/> <!-- 积分增益 -->
      <!-- 坐标系选择 -->
      <arg name="use_ned_frame" value="$(arg use_ned_frame)"/>
      <!-- 推力器管理相关配置 -->
      <arg name="thruster_manager_output_dir" value="$(arg thruster_manager_output_dir)"/>
      <arg name="thruster_manager_config_file" value="$(arg thruster_manager_config_file)"/>
      <arg name="tam_file" value="$(arg tam_file)"/>
    </include>

    <!-- 数据记录功能 -->
    <!-- Initialize the recording fo the simulation according to the record flag  -->
    <include file="$(find rexrov2_control)/launch/record_demo.launch">
      <arg name="record" value="$(arg record)"/> <!-- 记录开关 -->
      <arg name="use_ned_frame" value="$(arg use_ned_frame)"/> <!-- 坐标系选择 -->
      <arg name="uuv_name" value="$(arg uuv_name)"/> <!-- 机器人名称 -->
    </include>

    <!-- 遥操作功能 -->
    <group if="$(arg teleop_on)">
      <!-- Start joystick teleop node  -->
      <include file="$(find uuv_teleop)/launch/uuv_teleop.launch">
          <arg name="uuv_name" value="$(arg uuv_name)"/> <!-- 机器人命名空间 -->
          <arg name="joy_id" value="$(arg joy_id)"/> <!-- 手柄设备ID -->
          <arg name="output_topic" value="cmd_vel"/> <!-- 输出控制指令的话题 -->
          <arg name="message_type" value="twist"/> <!-- 消息类型：Twist速度指令 -->
          <!-- 手柄轴映射 -->
          <arg name="axis_yaw" value="$(arg axis_yaw)"/> <!-- 偏航轴映射 -->
          <arg name="axis_x" value="$(arg axis_x)"/> <!-- X轴映射 -->
          <arg name="axis_y" value="$(arg axis_y)"/> <!-- Y轴映射 -->
          <arg name="axis_z" value="$(arg axis_z)"/> <!-- Z轴映射 -->
          <!-- 控制增益设置 -->
          <arg name="gain_yaw" default="0.2"/> <!-- 偏航控制增益 -->
          <arg name="gain_x" default="0.5"/> <!-- X轴控制增益 -->
          <arg name="gain_y" default="0.5"/> <!-- Y轴控制增益 -->
          <arg name="gain_z" default="0.5"/> <!-- Z轴控制增益 -->
      </include>
    </group>

    <!-- 可视化功能 -->
    <group if="$(arg gui_on)">
      <!-- Start RViz  -->
      <node name="rviz" pkg="rviz" type="rviz" output="screen" args="-d $(find rexrov2_control)/rviz/rexrov2_control.rviz"/>
      <!-- 启动RViz并加载预设的RexROV2控制配置文件 -->
    </group>
</launch>

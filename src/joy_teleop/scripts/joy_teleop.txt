def joy_callback(self, data):
    for c in self.command_list:
        if self.match_command(c, data.buttons):  # 匹配按钮组合
            self.run_command(c, data)  # 执行对应命令

4. 命令匹配逻辑
python

def match_command(self, c, buttons):
    # 检查按钮组合是否匹配
    for b in self.command_list[c]['buttons']:
        if buttons[b] != 1:  # 按钮未按下
            return False
    # 检查是否允许多按钮组合
    if self.command_list[c]['allow_combinations']:
        return sum(buttons) >= len(self.command_list[c]['buttons'])
    else:
        return sum(buttons) == len(self.command_list[c]['buttons'])

6. 高级特性

动态类型加载：
python

def get_message_type(self, type_name):
    # 动态加载ROS消息类型，如"geometry_msgs/Twist"
    package, message = type_name.split('/')
    mod = importlib.import_module(package + '.msg')
    return getattr(mod, message)

异步服务调用：
python

class AsyncServiceProxy(object):
    # 在单独线程中调用服务，避免阻塞
    def __call__(self, request):
        self._thread = Thread(target=self._service_proxy, args=[request])
        self._thread.start()

自动重连：
python

def update_actions(self, evt=None):
    # 定期检查并重新连接离线的Action服务器
    for name, cmd in self.command_list.items():
        if cmd['action_name'] in self.offline_actions:
            self.register_action(name, cmd)

7. 配置示例（YAML格式）
yaml

teleop:
  cmd_vel:
    type: topic
    topic_name: /robot/cmd_vel
    message_type: geometry_msgs/Twist
    deadman_buttons: [4]  # L1按钮
    axis_mappings:
      - axis: 1    # 左摇杆上下
        target: linear.x
        scale: 0.5
      - axis: 0    # 左摇杆左右  
        target: angular.z
        scale: 1.0
  
  gripper_open:
    type: action
    action_name: /gripper_controller/gripper_action
    action_goal: {position: 0.1, max_effort: 50.0}
    buttons: [0]   # X按钮
